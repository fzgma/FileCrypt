# Ubuntu 22.04 的系统 CMake 为 3.22；复验不需要主项目的编译配置。
cmake_minimum_required(VERSION 3.16)
project(FileCryptRuntimeTests LANGUAGES NONE)
enable_testing()

# 只注册测试，复用已编译程序，不查找编译器或依赖库。
if(NOT FILECRYPT_BUILD_DIR)
    message(FATAL_ERROR "FILECRYPT_BUILD_DIR is required")
endif()
get_filename_component(FILECRYPT_BUILD_DIR "${FILECRYPT_BUILD_DIR}" ABSOLUTE)
set(test_suffix "")
if(WIN32)
    set(test_suffix ".exe")
endif()
foreach(test_target header_test metadata_test aad_test detect_test crypto_test compression_test directory_test expected_test)
    set(${test_target}_command "${FILECRYPT_BUILD_DIR}/tests/unit/${test_target}${test_suffix}")
endforeach()
foreach(test_target file_crypt_test directory_crypt_test console_test)
    set(${test_target}_command "${FILECRYPT_BUILD_DIR}/tests/integration/${test_target}${test_suffix}")
endforeach()
if(NOT FILECRYPT_CLI)
    set(FILECRYPT_CLI "${FILECRYPT_BUILD_DIR}/filecrypt${test_suffix}")
endif()
set(filecrypt_command "${FILECRYPT_CLI}")
get_filename_component(FILECRYPT_TEST_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
include("${FILECRYPT_TEST_SOURCE_DIR}/register_tests.cmake")
